var compiler = "gcc";
var flags = "-Wall -Wextra -O2";
var output = "test/proj/main";
var srcs = match "*.c";

task build {
  run "{compiler} {flags} {srcs} -o {output}";
}

task test {
  needs clean & build;
  run "./{output} --test";
}

task run {
  needs build;
  run "./{output}";
}

task clean {
  run "rm -f {output}";
}
